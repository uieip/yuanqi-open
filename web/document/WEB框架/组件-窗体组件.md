```javascript
export interface QueuePopperOptions {
    ajaxUrl?: string;
    titles?: any[];
    rows?: Record<string, any>[];
    keyId?: string;
    fkrows?: Record<string, any>[], // 关联数据

    containerRef?: HTMLElement | null, //调用控件
    top?: string | number,
    left?: string | number,
    width?: string | number,
    height?: string | number,
} 

// 排序组件（表格模式）
queue(options: QueuePopperOptions, callBack: (rows: Record<string, any>[]) => void) {
    if (!options || !options.containerRef) {
        console.error("containerRef 不能为空")
        return
    }
    if (!callBack) {
        console.error("callBack 不能为空")
        return
    }

    let parent = true;
    if (!window.parent || window.parent === window) {
        parent = false;
    }

    let dialogId: string = "";
    const oldRows = ref<Record<string, any>[]>([]);
    const params = ref({
        btn_disabled: true,
    });

    const rect = options.containerRef?.getBoundingClientRect() || { top: 0, left: 0, width: 500, height: 350 };
    // 创建完整内容，包括按钮
    const content = {
        h: 'div',
        props: {
            style: {
                height: '100%',
                width: '100%',
                padding: '5px 6px 4px 6px',
                backgroundColor: '#f5f7fa',
                overflow: 'auto'
            }
        },
        children: {
            h: "YqQueue",
            props: {
                ajaxUrl: options.ajaxUrl || '',
                titles: options.titles || [],
                rows: options.rows || [],
                keyId: options.keyId || '',
                fkrows: options.fkrows || [],
                height: yqUiPub.convertWindowValue(options.height) - 65,
                'onUpdate:rows': (val: Record<string, any>[]) => {
                    //console.log('onUpdate:rows val:', val);
                    params.value.btn_disabled = false;
                    oldRows.value = val;

                    YqWindowApi.setParams(dialogId, params.value, parent);
                }
            }
        }
    };

    //console.log('height:', yqUiPub.convertWindowValue(options.height));

    dialogId = YqWindowApi.show({
        containerRef: options.containerRef || null,
        placement: "popper",
        title: "排序设置",
        icon: '/icon/远齐.jpg',
        showTitle: false,
        showClose: false,
        type: "param",
        content: content,
        top: options.top || 0,
        left: options.left || 0,
        width: options.width || rect.width,
        height: options.height || rect.height,
        parent: true,
        fullscreen: false,
        showFooter: true,
        children: {
            h: 'div',
            props: {
                style: {
                    display: 'flex',
                    justifyContent: 'flex-end',
                    gap: '8px',
                    marginRight: '10px'
                }
            },
            children: [{
                h: 'ElButton',
                props: {
                    type: 'primary',
                    size: 'default',
                    disabled: (windowState: Record<string, any>) => {
                        return windowState?.btn_disabled ?? params.value.btn_disabled;
                    },
                    onClick: () => {
                        callBack(oldRows.value);
                        YqWindowApi.close(dialogId, parent);
                    },
                },
                children: '确定'
            }, {
                h: 'ElButton',
                props: {
                    type: 'default',
                    size: 'default',
                    onClick: () => {
                        YqWindowApi.close(dialogId, parent);
                    }
                },
                children: '取消'
            }]
        },
    });

    YqWindowApi.setParams(dialogId, params.value, parent);

    // 添加参数变化监听器
    YqWindowApi.addParamsListener(dialogId, (newParams) => {
        console.log('参数变化:', newParams);
        params.value.btn_disabled = newParams.disabled;
    }, parent);
}
```

# 二、YqMessageBox

## 1.消息框

```javascript
// 扩展 MessageBoxOptions 接口，增加 el 属性
export interface YqMessageBoxOptions extends ElMessageBoxOptions {
    el?: Element;
    top?: number;
    left?: number;
    width?: number;
    height?: number;
    /* 多语言字典
    示例
    {
        "zh-CN": {
            "title": "添加",
            "message": "添加新记录",
            "cancelButtonText": "取消",
            "confirmButtonText": "确认"
        },
        "en": {
            "title": "Add",
            "message": "Add new record",
            "cancelButtonText": "Cancel",
            "confirmButtonText": "Confirm"
        }
    }
    */
    labelDict?: Record<string, {
        title?: string,
        message?: string,
        cancelButtonText?: string,
        confirmButtonText?: string,
    }>;
}

export class YqMessageBox {
    // 确认对话框
    static confirm(options: string | YqMessageBoxOptions): Promise<any>;
}

export default YqMessageBox;
```
- **说明**：`confirm` 方法参数可以是字符串或 `YqMessageBoxOptions` 对象。对象形式下，`labelDict` 翻译优先，其次 `options` 自带值，最后回退到默认值（取消/确认）。`type` 默认为 `'warning'`，对应不同的图标与颜色。
- 使用示例

```javascript
// 显示确认对话框
const showConfirm = () => {
    YqMessageBox.confirm({
        message: '您确定要执行此操作吗？',
        title: '确认操作',
        type: 'warning'
    }).then(() => {
        console.log('用户点击了确定');
    }).catch(() => {
        console.log('用户点击了取消');
    });
}
// 显示在指定按钮下方的确认对话框
const showTargetedConfirm = () => {
    YqMessageBox.confirm({
        message: '您确定要执行此操作吗？',
        title: '确认操作',
        type: 'warning',
        el: btn1.value.$el // 使用 $el 获取 DOM 元素
    }).then(() => {
        console.log('用户点击了确定');
    }).catch(() => {
        console.log('用户点击了取消');
    });
}
//或
const showTargetedConfirm = async () => {
    try {

            title: '确认操作',
```

# 六、YqMessage
全局消息提示工具，包装 ElementPlus `ElMessage`，支持 `labelDict` 多语言翻译。

```javascript
export interface YqMessageOptions extends Omit<MessageOptions, 'message'> {
    message?: string;
    /* 多语言消息字典
    示例
    {
        "zh-CN": { "message": "数据保存成功" },
        "en":    { "message": "Data saved successfully" }
    }
    */
    labelDict?: Record<string, { message?: string }>;
}

export type YqMessageParams = string | YqMessageOptions;

export type YqMessageHandler = ReturnType<typeof ElMessage>;

export interface YqMessageFn {
    (options?: YqMessageParams): YqMessageHandler;
    success(options?: YqMessageParams): YqMessageHandler;
    warning(options?: YqMessageParams): YqMessageHandler;
    info(options?: YqMessageParams): YqMessageHandler;
    error(options?: YqMessageParams): YqMessageHandler;
}

export const YqMessage: YqMessageFn;
export default YqMessage;
```

- **说明**：
  - 参数支持字符串或 `YqMessageOptions` 对象；字符串会自动转换为 `{ message: string }`。
  - 渲染时优先使用 `labelDict`（按当前 locale 解析）的 `message`，其次使用 `options.message`。
  - 调用方式与 `ElMessage` 一致：`YqMessage(...)`、`YqMessage.success(...)`、`YqMessage.warning(...)`、`YqMessage.info(...)`、`YqMessage.error(...)`。

- 使用示例

```javascript
// 字符串形式
YqMessage.success('保存成功');

// 对象形式 + 多语言
YqMessage.success({
    labelDict: {
        'zh-CN': { message: '数据保存成功' },
        'en':    { message: 'Data saved successfully' }
    }
});

// 对象形式（含 ElMessage 原生参数）
YqMessage.error({
    message: '保存失败',
    duration: 5000,
    showClose: true
});
```
