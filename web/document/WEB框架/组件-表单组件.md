``` javascript
const props = withDefaults(defineProps<{
  code?: string;
  lineNumbers?: boolean;
  isEdit?: boolean;
  mode?: 'text' | 'txt' | 'python' | 'javascript' | 'js' | 'html' | 'css' | 'json' | 'go' | 'golang' | 'csharp' | 'c#' | 'cs' | any;
  types?: Array<string>;
  indentSize?: number;
  autoCompletion?: boolean;
  autoFold?: boolean;
  highlightActiveLine?: boolean;
  tabSize?: number;
  wordWrap?: boolean;           // 是否启用自动换行

  setting?: boolean;
  clear?: boolean;                 // 是否清空代码
}>(), {
  code: '',                  // 初始代码为空
  lineNumbers: true,         // 开启行号显示
  isEdit: true,           // 初始为可编辑状态
  mode: 'json',            // 初始模式为python
  types: () => ['json', 'python', 'javascript', 'js', 'html', 'css', 'go', 'golang', 'csharp', 'c#', 'cs'],
  indentSize: 2,             // 设置缩进宽度为2个空格
  autoCompletion: true,      // 开启自动完成
  autoFold: true,            // 开启自动折叠
  highlightActiveLine: true, // 开启高亮当前行
  tabSize: 2,                // 设置Tab键宽度为2个空格
  wordWrap: true,           // 默认启用自动换行

  setting: true,
  clear: true,                 // 是否清空代码
});
```

- **事件**
``` javascript

  'change': [value: string];
  'update:code': [value: string];

```
