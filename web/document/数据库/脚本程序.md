数据库脚本程序可以替代数据库端存储过程程序。

#一、概述
程序默认输入属性参数
- _dbName：数据库
- _tbName：实际表名称
- _vName：逻辑表名称

page_callback： 为回调函数
##1.1 入口函数（action）
文件：demo/ab_test_m.js
```javascript
include("demo/ab_test_m_include.js");

//page_callback 为回调函数
function action(jsons, userid, username) {
    try {
        var rows = [];
        run(0, function () {
            var back = {
                error: 0,
                info: "success",
                rows: rows,
            };
            console.log("callback", back);

            page_callback(back);
        });
    }
    catch (e) {
        var back = {
            error: 9,
            info: e.message,
            rows: [],
        };
        console.log(e);

        page_callback(back);
    }

    function run(index, callback) {
        if (index >= jsons.length) {
            callback(rows);
            return;
        }

        var json = jsons[index];
        if (json.action == "add" || json.action == "upp" || json.action == "del") {
            test_fun[json.action](json.row, userid, username, function (res) {
                if (res.error == 0) {
                    rows.push(res.rows[0]);
                } else {
                    rows.push(res);
                }

                run(index + 1, callback);
            });
        } else {
            var res = data.action(_dbName, _vName, [json]);
            if (res.error == 0) {
                rows.push(res.rows[0]);
            } else {
                rows.push(res);
            }

            run(index + 1, callback);
        }
    }
}
```
文件：demo/ab_test_m_include.js
```javascript

var test_fun = {
    add: function (row, userid, username, callback) {
        row.c_userid = userid;
        row.c_username = username;
        row.createtime = test_fun.formatDate(new Date());
        row.m_userid = userid;
        row.m_username = username;
        row.modifytime = row.createtime;

        var res = data.action(_dbName, _vName, [{
            action: "add",
            row: row
        }]);

        callback(res);
    },
    upp: function (row, userid, username, callback) {
        row.m_userid = userid;
        row.m_username = username;
        row.modifytime = test_fun.formatDate(new Date());
        var res = data.action(_dbName, _vName, [{
            action: "upp",
            row: row
        }]);

        callback(res);
    },
    del: function (row, userid, username, callback) {
        var res = data.action(_dbName, _vName, [{
            action: "del",
            row: row
        }]);

        callback(res);
    },
    state: function (row, userid, username, callback) {
        row.m_userid = userid;
        row.m_username = username;
        row.modifytime = test_fun.formatDate(new Date());
        var res = data.action(_dbName, _vName, [{
            action: "upp_",
            row: row
        }]);

        if (typeof callback == "function") {
            callback(res);
        }

        if (typeof page_callback == "function") {
            page_callback(res);
        }
    },
    formatDate: function (date) {
        const year = date.getFullYear();
        const month = String(date.getMonth() + 1).padStart(2, '0');
        const day = String(date.getDate()).padStart(2, '0');
        const hours = String(date.getHours()).padStart(2, '0');
        const minutes = String(date.getMinutes()).padStart(2, '0');
        const seconds = String(date.getSeconds()).padStart(2, '0');

        return `${year}-${month}-${day} ${hours}:${minutes}:${seconds}`;
    }
}
```

#二、系统函数
##2.1 加载系统扩展函数
- require(module)
- module: 模块名称或模块列表, 字符串或字符串数组

## 2.2 包含js文件引用
- include({file1, file2,...})
- 包含文件，可以是相对路径或绝对路径，文件不存在时会报错。

##2.3 数据库函数（data.*）
- 同脚本语言数据库操作，见《页面脚本语言》->数据库操作。
- action：直接执行数据库内部操作，页面脚本语言执行数据库标准操作，如有存储过程或脚本，执行存储过程或脚本。
