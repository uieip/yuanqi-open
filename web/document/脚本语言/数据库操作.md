### 9)query  执行sql查询语句
var res=data.query(dbName,sqlStr,params)
- sqlStr格式：select * from ab_test_m where sid = $1
- params格式：["param1"]
- 返回格式：{error:0,info:"success",rows:[{sid:"sid1",textbox:"textbox1"},{sid:"sid2",textbox:"textbox2"}]}

```javascript
var sqlStr = "select * from ab_test_m order by textbox where sid=$1";
var params = ["29251c7d-6cb8-466a-b2da-02692fd2a2b8"];
var result = data.query(dbName,sqlStr, params);
if (result.error!= "0") {
    return result;
}
```

### 10)exec 执行sql语句
var res=data.exec(dbName,sqlStr,params)；
如：insert、update、delete等操作
- sqlStr格式：update ab_test_m set textbox = $1 where sid = $2
- params格式：["param1","param2",...]
- 返回格式：{error:0,info:"success"}
```javascript
function action(jsons, userid, username) {
    var sqlStr = "update ab_test_m set textbox = $1 where sid = $2";
    var params = ["test", "9bae3764-cbbe-456e-b86f-9f62d80ffe75"];
    var result = data.exec(dbName,sqlStr, params);
    if (result.error!= "0") {
        return result;
    }
    
	return {
          error: 0,
          info: "ok"
     };
} 
```

### 11)执行标准action
```javascript
var back = data.action(dbName,tbName,jsons)；
```
- demo
```javascript
function action(jsons, userid, username) {
    var back = data.action(dbName,vName,jsons);
    if (back.error!= "0") {
        return {
            error:  back.error,
            info: back.info,
        };
    }else{
        return {
            error: 0,
            info: back.info,
            rows: back.rows,
        };
    }
} 
```
- jsons格式示例
```json
[{
	action:"add",
	row:{
		name:"name1",
		code:"code1"
		}
  },{
   action:"upp",
   row:{
		sid:"sid2",
		name:"name2",
		code:"code2"
		}
}]
```
- 系统标准数据库操作

| no. | action  | 说明  |
| --- | ------- | ----- |
|  1  | add | 添加  |
|  2  | upp | 修改,row按输入修改,未包含在row字段数据清空 |
|  3  | upp_ | 修改,row输入修改,未包含在row字段不修改 |
|  4  | del | 删除 |
|  5  | move | 移动 |
|  6  | paste | 移动 |

- 返回格式
```json
{
	error:0,
	info:"success",
	rows:[{
			error:0,
			info:"success",
			action:"add",
			sid:"sid1"
		},{
			error:0,
			info:"success",
			action:"upp",
			sid:"sid2"
		}]
}
```
